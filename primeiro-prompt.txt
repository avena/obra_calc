### 📝 Resumo das Fórmulas e Macetes (Texto)

**1. Alvenaria (Blocos)**
*   **Regra de Ouro:** Sempre desconte portas e janelas da área total.
*   **Rendimento dos Blocos:**
    *   Bloco 9x19x19 = **25 un/m²**
    *   Bloco 9x19x29 = **16,7 un/m²**
    *   Bloco 9x19x39 = **12,5 un/m²**
*   **Perda:** Adicione sempre **5%** a mais na compra.
*   **Macete do Milheiro:** Um pallet de bloco 19cm rende ~40m²; 29cm rende ~60m²; 39cm rende ~80m².

**2. Reboco e Argamassa**
*   **Fórmula:** `Volume = Área × Espessura`
*   **Rendimento por m³ de argamassa:**
    *   Espessura 1 cm = 100 m²
    *   Espessura 1,5 cm = 66,7 m²
    *   Espessura 2 cm = 50 m²
    *   Espessura 2,5 cm = 40 m²
*   **Macete:** 1 m³ de argamassa equivale a aproximadamente 56 latas de 18L.

**3. Piso e Porcelanato**
*   **Fórmula:** `Quantidade = Área do ambiente ÷ Área da peça`
*   **Perdas (sempre arredonde para cima):**
    *   Paginação simples (reta): +5% a 10%
    *   Paginação diagonal ou muitos recortes: +10% a 15%

**4. Rodapé e Contrapiso**
*   **Rodapé:** Calcule o perímetro do ambiente (soma de todos os lados) e subtraia a largura das portas.
*   **Contrapiso:** `Volume = Área × Espessura`. Adicione 10% de folga.

**5. Concreto**
*   **Fórmula geral:** `Volume = Base × Altura × Comprimento`
*   **1 m³ de concreto rende:**
    *   Laje de 5 cm = 20 m²
    *   Laje de 8 cm = 12,5 m²
    *   Laje de 10 cm = 10 m²
    *   Laje de 12 cm = 8,3 m²
*   **Peso próprio do concreto:** 25 kN/m³ (≈ 2,55 tf/m³).

**6. Aço**
*   **Peso Linear:** `Peso (kg/m) ≈ (Diâmetro²) ÷ 162`. Ex: Barra de 10mm = 100/162 = 0,617 kg/m.
*   **Quantidade de barras:** `N = (Comprimento ÷ Espaçamento) + 1`

    <script>
        function openTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(t => t.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            
            document.getElementById(tabId).classList.remove('hidden');
            event.currentTarget.classList.add('active');
            
            // Trigger calculations
            if(tabId === 'alvenaria') calcAlvenaria();
            if(tabId === 'piso') calcPiso();
            if(tabId === 'argamassa') calcArgamassa();
            if(tabId === 'concreto') calcConcreto();
            if(tabId === 'aco') calcAco();
        }

        function calcAlvenaria() {
            const comp = parseFloat(document.getElementById('alv_comp').value) || 0;
            const alt = parseFloat(document.getElementById('alv_alt').value) || 0;
            const porta = parseFloat(document.getElementById('alv_porta').value) || 0;
            const janela = parseFloat(document.getElementById('alv_janela').value) || 0;
            const rendimento = parseFloat(document.getElementById('alv_tipo').value) || 25;

            const areaTotal = comp * alt;
            const areaLiquida = Math.max(0, areaTotal - porta - janela);
            const blocos = Math.ceil(areaLiquida * rendimento);
            const blocosComPerda = Math.ceil(blocos * 1.05);

            document.getElementById('alv_res').innerText = `${blocosComPerda} blocos`;
        }

        function calcPiso() {
            const area = parseFloat(document.getElementById('pis_area').value) || 0;
            const larg = parseFloat(document.getElementById('pis_larg').value) || 1;
            const comp = parseFloat(document.getElementById('pis_comp').value) || 1;
            const perda = parseFloat(document.getElementById('pis_perda').value) || 0;

            const areaPeca = (larg * comp) / 10000;
            if(areaPeca === 0) return;
            
            let qtd = area / areaPeca;
            let qtdArred = Math.ceil(qtd);
            let qtdFinal = Math.ceil(qtdArred * (1 + (perda/100)));

            document.getElementById('pis_res').innerText = `${qtdFinal} peças (${qtd.toFixed(2)} exatas)`;
        }

        function calcArgamassa() {
            const area = parseFloat(document.getElementById('arg_area').value) || 0;
            const esp = parseFloat(document.getElementById('arg_esp').value) || 0;

            const volume = area * (esp / 100);
            const latas = volume * 56;

            document.getElementById('arg_res').innerText = `${volume.toFixed(2)} m³ (≈ ${Math.ceil(latas)} latas de 18L)`;
        }

        function calcConcreto() {
            const comp = parseFloat(document.getElementById('conc_comp').value) || 0;
            const larg = parseFloat(document.getElementById('conc_larg').value) || 0;
            const alt = parseFloat(document.getElementById('conc_alt').value) || 0;

            const volume = comp * larg * alt;
            document.getElementById('conc_res').innerText = `${volume.toFixed(3)} m³`;
        }

        function calcAco() {
            const diam = parseFloat(document.getElementById('aco_diam').value) || 0;
            const comp = parseFloat(document.getElementById('aco_comp').value) || 0;
            const esp = parseFloat(document.getElementById('aco_esp').value) || 20;
            const laje = parseFloat(document.getElementById('aco_laje').value) || 1;

            const pesoM = (diam * diam) / 162;
            const pesoTotal = pesoM * comp;
            document.getElementById('aco_peso').innerText = `${pesoTotal.toFixed(2)} kg (${pesoM.toFixed(3)} kg/m)`;

            const barras = Math.ceil((laje / (esp / 100))) + 1;
            document.getElementById('aco_barras').innerText = `${barras} barras`;
        }

        // Initialize
        calcAlvenaria();
        calcPiso();
        calcArgamassa();
        calcConcreto();
        calcAco();
    </script>
